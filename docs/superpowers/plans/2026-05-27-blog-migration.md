# Blog Migration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Migrate 25 old static blog posts from `tangwz/tangwz.github.io` into maintainable Astro Markdown posts that render with the current blog styling only.

**Architecture:** Build a repeatable Node-based migration tool under `scripts/migrate-old-blog/`. The tool parses old static HTML, extracts metadata, converts only semantic article content to Markdown, copies original local assets into each post folder, writes a migration report, and fails on blocking anomalies.

**Tech Stack:** Astro content collections, Node.js ESM, `cheerio@1.2.0`, `turndown@7.2.4`, `node:test`, `pnpm`, `npm run build`.

---

## File Structure

- Create `scripts/migrate-old-blog/lib/metadata.mjs`: extract frontmatter metadata from JSON-LD, Open Graph meta tags, and `index.json` fallback data.
- Create `scripts/migrate-old-blog/lib/assets.mjs`: resolve old image URLs to original source files and copy them into post-local `assets/` directories.
- Create `scripts/migrate-old-blog/lib/content.mjs`: clean old theme HTML and convert article content to Markdown without old style classes, wrappers, inline styles, or `srcset`.
- Create `scripts/migrate-old-blog/migrate.mjs`: CLI orchestrator for source discovery, conversion, output writing, and report generation.
- Create focused tests in `scripts/migrate-old-blog/*.test.mjs`.
- Modify `package.json`: add migration scripts.
- Modify `pnpm-lock.yaml`: generated by `pnpm add -D cheerio@1.2.0 turndown@7.2.4`.
- Modify `.gitignore`: ignore local old-site clone cache and temporary migration reports.
- Generate `src/content/posts/<slug>/index.md` and `src/content/posts/<slug>/assets/*` during execution.

function cleanText(value) {
  return String(value ?? "")
    .replace(/\s+/g, " ")
    .trim();
}

function metaContent($, selector) {
  return cleanText($(selector).first().attr("content"));
}

function canonicalUrl($) {
  return cleanText($('link[rel="canonical"]').first().attr("href"));
}

function firstArticleJsonLd($) {
  const scripts = $('script[type="application/ld+json"]').toArray();
  for (const script of scripts) {
    const raw = $(script).text();
    if (!raw.trim()) continue;
    try {
      const parsed = JSON.parse(raw);
      const entries = Array.isArray(parsed) ? parsed : [parsed];
      const article = entries.find(entry => entry?.["@type"] === "Article");
      if (article) return article;
    } catch {
      continue;
    }
  }
  return {};
}

function normalizeTags(value) {
  if (Array.isArray(value)) {
    return value.map(cleanText).filter(Boolean);
  }
  if (typeof value === "string" && value.trim()) {
    return value
      .split(",")
      .map(cleanText)
      .filter(Boolean);
  }
  return [];
}

function htmlTitle($) {
  return cleanText($("title").first().text()).replace(/\s+(?:\u00b7|-)\s+.*$/, "");
}

function dateOnlyToIso(date) {
  if (/^\d{4}-\d{2}-\d{2}$/.test(date)) {
    return `${date}T00:00:00+00:00`;
  }
  return date;
}

function yamlString(value) {
  return JSON.stringify(String(value ?? ""));
}

export function extractMetadata($, indexRecords, permalink) {
  const article = firstArticleJsonLd($);
  const fallback = indexRecords.get(permalink) ?? {};
  const fallbacks = [];

  let title = cleanText(article.headline ?? article.name) || metaContent($, 'meta[property="og:title"]');
  if (!title) {
    title = cleanText(fallback.title) || htmlTitle($);
    fallbacks.push("title");
  }

  let pubDatetime = cleanText(article.datePublished) || metaContent($, 'meta[property="article:published_time"]');
  if (!pubDatetime && fallback.date) {
    pubDatetime = dateOnlyToIso(cleanText(fallback.date));
    fallbacks.push("pubDatetime");
  }

  const modDatetime =
    cleanText(article.dateModified) || metaContent($, 'meta[property="article:modified_time"]');

  let description =
    metaContent($, 'meta[property="og:description"]') ||
    cleanText(article.abstract) ||
    metaContent($, 'meta[name="description"]');
  if (!description && fallback.summary) {
    description = cleanText(fallback.summary);
    fallbacks.push("description");
  }

  let tags = normalizeTags(article.keywords);
  if (tags.length === 0) {
    tags = ["others"];
    fallbacks.push("tags");
  }

  const canonicalURL = canonicalUrl($) || `https://tangwz.com${permalink}`;
  const ogImage = metaContent($, 'meta[property="og:image"]') || metaContent($, 'meta[name="twitter:image"]');

  return {
    title,
    pubDatetime,
    ...(modDatetime ? { modDatetime } : {}),
    description,
    tags,
    canonicalURL,
    ...(ogImage ? { ogImage } : {}),
    draft: false,
    fallbacks,
  };
}

export function toFrontmatter(metadata) {
  const lines = [
    "---",
    `title: ${yamlString(metadata.title)}`,
    `pubDatetime: ${metadata.pubDatetime}`,
  ];

  if (metadata.modDatetime) lines.push(`modDatetime: ${metadata.modDatetime}`);

  lines.push(`description: ${yamlString(metadata.description)}`);
  lines.push("tags:");
  for (const tag of metadata.tags) {
    lines.push(`  - ${yamlString(tag)}`);
  }

  if (metadata.canonicalURL) lines.push(`canonicalURL: ${yamlString(metadata.canonicalURL)}`);
  if (metadata.ogImage) lines.push(`ogImage: ${yamlString(metadata.ogImage)}`);
  lines.push("draft: false");
  lines.push("---", "");
  return lines.join("\n");
}
```

import { copyFile, mkdir, stat } from "node:fs/promises";
import path from "node:path";

const DERIVATIVE_PATTERN = /_hu[0-9a-f]+_\d+_\d+x0_resize(?:_[^.]+)?(\.[^.]+)$/i;

export function isResizeDerivative(fileName) {
  return DERIVATIVE_PATTERN.test(fileName);
}

export function originalNameForDerivative(fileName) {
  return fileName.replace(/_hu[0-9a-f]+_\d+_\d+x0_resize(?:_[^.]+)?(\.[^.]+)$/i, "$1");
}

function normalizeImagePath(imageUrl) {
  const url = String(imageUrl ?? "").trim();
  if (!url) return "";
  try {
    const parsed = new URL(url, "https://tangwz.com");
    return decodeURIComponent(parsed.pathname.replace(/^\/+/, ""));
  } catch {
    return decodeURIComponent(url.replace(/^\/+/, ""));
  }
}

async function exists(filePath) {
  try {
    const info = await stat(filePath);
    return info.isFile();
  } catch {
    return false;
  }
}

function uniqueOutputName(usedNames, fileName) {
  const parsed = path.parse(fileName);
  let candidate = fileName;
  let index = 2;
  while (usedNames.has(candidate)) {
    candidate = `${parsed.name}-${index}${parsed.ext}`;
    index += 1;
  }
  usedNames.add(candidate);
  return candidate;
}

export async function copyReferencedAsset({
  sourceRoot,
  outputPostDir,
  postSlug,
  imageUrl,
  usedNames = new Set(),
  copiedBySource = new Map(),
}) {
  const normalized = normalizeImagePath(imageUrl);
  if (!normalized) return { markdownPath: "", missing: imageUrl };

  const sourceRelative = normalized.startsWith(`posts/${postSlug}/`)
    ? normalized
    : path.posix.join("posts", postSlug, path.posix.basename(normalized));

  const fileName = path.posix.basename(sourceRelative);
  const originalFileName = isResizeDerivative(fileName) ? originalNameForDerivative(fileName) : fileName;
  const preferredPath = path.join(sourceRoot, "posts", postSlug, originalFileName);
  const fallbackPath = path.join(sourceRoot, sourceRelative);
  const sourcePath = (await exists(preferredPath)) ? preferredPath : fallbackPath;

  if (!(await exists(sourcePath))) {
    return { markdownPath: "", missing: imageUrl };
  }

  if (copiedBySource.has(sourcePath)) {
    return { ...copiedBySource.get(sourcePath), reused: true };
  }

  const outputName = uniqueOutputName(usedNames, path.basename(sourcePath));
  const outputAssetsDir = path.join(outputPostDir, "assets");
  await mkdir(outputAssetsDir, { recursive: true });
  const outputPath = path.join(outputAssetsDir, outputName);
  await copyFile(sourcePath, outputPath);

  const result = {
    markdownPath: `./assets/${outputName}`,
    copiedFrom: sourcePath,
    copiedTo: outputPath,
  };
  copiedBySource.set(sourcePath, result);
  return result;
}
```

import TurndownService from "turndown";

const BODY_SELECTOR = "article .min-w-0.min-h-0.max-w-prose, article .max-w-prose";

function removeOldThemeAttributes($, root) {
  $(root)
    .find("*")
    .each((_, element) => {
      const node = $(element);
      node.removeAttr("class");
      node.removeAttr("style");
      node.removeAttr("srcset");
      node.removeAttr("sizes");
      node.removeAttr("loading");
      node.removeAttr("decoding");
      node.removeAttr("width");
      node.removeAttr("height");
    });
}

function removeGeneratedNodes($, root) {
  $(root).find("script, style, footer, nav, details, .toc, .anchor").remove();
  $(root).find("span").each((_, element) => {
    const text = $(element).text().trim();
    if (text === "#" || $(element).find("a[aria-label]").length > 0) {
      $(element).remove();
    }
  });
  $(root)
    .find("a[aria-label]")
    .filter((_, element) => $(element).text().trim() === "#" || ($(element).attr("href") ?? "").startsWith("#"))
    .remove();
}

function unwrapThemeOnlyNodes($, root) {
  $(root).find("figure").each((_, element) => {
    const image = $(element).find("img").first();
    if (image.length > 0) {
      $(element).replaceWith(image);
    }
  });
}

export function extractArticleHtml($) {
  const body = $(BODY_SELECTOR).last();
  const root = body.length > 0 ? body.clone() : $("article").clone();

  removeGeneratedNodes($, root);
  unwrapThemeOnlyNodes($, root);
  removeOldThemeAttributes($, root);

  return root.html()?.trim() ?? "";
}

export function htmlToMarkdown(html) {
  const turndown = new TurndownService({
    headingStyle: "atx",
    codeBlockStyle: "fenced",
    bulletListMarker: "-",
    emDelimiter: "_",
    strongDelimiter: "**",
  });

  turndown.addRule("fencedCode", {
    filter: node => node.nodeName === "PRE" && node.firstChild?.nodeName === "CODE",
    replacement: (_content, node) => {
      const code = node.textContent.replace(/\n+$/, "");
      return `\n\n\`\`\`\n${code}\n\`\`\`\n\n`;
    },
  });

  return turndown
    .turndown(html)
    .replace(/\n{3,}/g, "\n\n")
    .trim()
    .concat("\n");
}

```js
import { load } from "cheerio";
import { mkdir, readFile, readdir, rm, writeFile } from "node:fs/promises";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { copyReferencedAsset } from "./lib/assets.mjs";
import { extractArticleHtml, htmlToMarkdown } from "./lib/content.mjs";
import { extractMetadata, toFrontmatter } from "./lib/metadata.mjs";

function parseArgs(argv) {
  const args = new Map();
  for (let index = 0; index < argv.length; index += 2) {
    args.set(argv[index], argv[index + 1]);
  }
  return {
    sourceRoot: args.get("--source"),
    outRoot: args.get("--out") ?? "src/content/posts",
    reportPath: args.get("--report") ?? "migration-report.json",
    expectedCount: Number(args.get("--expected-count") ?? 25),
  };
}

async function readIndexRecords(sourceRoot) {
  try {
    const raw = await readFile(path.join(sourceRoot, "index.json"), "utf8");
    const records = JSON.parse(raw);
    return new Map(records.map(record => [record.permalink, record]));
  } catch {
    return new Map();
  }
}

async function discoverPosts(sourceRoot) {
  const postsRoot = path.join(sourceRoot, "posts");
  const entries = await readdir(postsRoot, { withFileTypes: true });
  return entries
    .filter(entry => entry.isDirectory() && /^20\d{4}-/.test(entry.name))
    .map(entry => ({
      slug: entry.name,
      htmlPath: path.join(postsRoot, entry.name, "index.html"),
      permalink: `/posts/${entry.name}/`,
    }))
    .sort((a, b) => a.slug.localeCompare(b.slug));
}

function imageUrlsFromMarkdown(markdown) {
  return [...markdown.matchAll(/!\[[^\]]*\]\(([^)]+)\)/g)].map(match => match[1]);
}

function hasLocalImageMissing(markdown) {
  return imageUrlsFromMarkdown(markdown).some(url => url.startsWith("/posts/") || url.startsWith("https://tangwz.com/posts/"));
}

async function convertPost({ sourceRoot, outRoot, indexRecords, post }) {
  const html = await readFile(post.htmlPath, "utf8");
  const $ = load(html, { decodeEntities: true });
  const metadata = extractMetadata($, indexRecords, post.permalink);
  const outputPostDir = path.join(outRoot, post.slug);
  const usedNames = new Set();
  const copiedBySource = new Map();
  const copiedAssets = [];
  const missingAssets = [];

  let articleHtml = extractArticleHtml($);
  await rm(outputPostDir, { recursive: true, force: true });
  await mkdir(outputPostDir, { recursive: true });

  const articleDom = load(articleHtml, { decodeEntities: true }, false);
  const imageNodes = articleDom("img").toArray();

  for (const image of imageNodes) {
    const node = articleDom(image);
    const source = node.attr("src");
    const result = await copyReferencedAsset({
      sourceRoot,
      outputPostDir,
      postSlug: post.slug,
      imageUrl: source,
      usedNames,
      copiedBySource,
    });
    if (result.markdownPath) {
      node.attr("src", result.markdownPath);
      copiedAssets.push(result);
    } else if (result.missing) {
      missingAssets.push(result.missing);
    }
  }

  if (metadata.ogImage) {
    const ogResult = await copyReferencedAsset({
      sourceRoot,
      outputPostDir,
      postSlug: post.slug,
      imageUrl: metadata.ogImage,
      usedNames,
      copiedBySource,
    });
    if (ogResult.markdownPath) {
      metadata.ogImage = ogResult.markdownPath;
      copiedAssets.push(ogResult);
    }
  }

  articleHtml = articleDom.root().html();
  const markdownBody = htmlToMarkdown(articleHtml);
  const markdown = `${toFrontmatter(metadata)}${markdownBody}`;

  await writeFile(path.join(outputPostDir, "index.md"), markdown);

  const residualHtml = /<\/?[a-z][\s\S]*>/i.test(markdownBody);
  const textLength = markdownBody.replace(/!\[[^\]]*\]\([^)]+\)/g, "").replace(/\s+/g, " ").trim().length;
  const blockers = [];
  if (!metadata.title || !metadata.pubDatetime || !metadata.description) blockers.push("missing required metadata");
  if (textLength < 50) blockers.push("body too short");
  if (missingAssets.length > 0) blockers.push("missing local assets");
  if (hasLocalImageMissing(markdown)) blockers.push("unmigrated local image urls");

  return {
    slug: post.slug,
    outputPath: path.join(outputPostDir, "index.md"),
    copiedAssets,
    missingAssets,
    fallbacks: metadata.fallbacks,
    residualHtml,
    textLength,
    imageCount: imageNodes.length,
    codeBlockCount: (markdownBody.match(/```/g) ?? []).length / 2,
    blockers,
  };
}

export async function runMigration({ sourceRoot, outRoot, reportPath, expectedCount = 25 }) {
  if (!sourceRoot) throw new Error("Missing --source");

  const indexRecords = await readIndexRecords(sourceRoot);
  const posts = await discoverPosts(sourceRoot);
  const results = [];
  const blockers = [];

  if (posts.length !== expectedCount) {
    blockers.push(`expected ${expectedCount} posts but found ${posts.length}`);
  }

  for (const post of posts) {
    const result = await convertPost({ sourceRoot, outRoot, indexRecords, post });
    results.push(result);
    for (const blocker of result.blockers) {
      blockers.push(`${post.slug}: ${blocker}`);
    }
  }

  const report = {
    generatedAt: new Date().toISOString(),
    sourceRoot,
    outRoot,
    expectedCount,
    migratedCount: posts.length,
    blockers,
    posts: results,
  };

  await writeFile(reportPath, JSON.stringify(report, null, 2));
  return report;
}

async function main() {
  const options = parseArgs(process.argv.slice(2));
  const report = await runMigration(options);
  console.log(`Migrated ${report.migratedCount} posts`);
  console.log(`Blockers: ${report.blockers.length}`);
  if (report.blockers.length > 0) {
    for (const blocker of report.blockers) console.error(`- ${blocker}`);
    process.exitCode = 1;
  }
}

const isMain = process.argv[1] === fileURLToPath(import.meta.url);
if (isMain) {
  main().catch(error => {
    console.error(error);
    process.exitCode = 1;
  });
}
```
